async function fetchMovideskTicket() { const token = document.getElementById('movideskToken').value.trim(); const ticketId = document.getElementById('movideskTicketId').value.trim(); if (!token || !ticketId) { alert('Por favor, informe a Chave API do Movidesk e o Número do Chamado.'); return; } const statusEl = document.getElementById('movideskStatus'); statusEl.innerHTML = '⌛ Conectando ao Movidesk...'; // URL oficial da API OData do Movidesk const rawUrl = `https://api.movidesk.com/public/v1/tickets?token=${token}&$filter=id eq ${ticketId}&$expand=owner,clients,actions`; // Utiliza proxy CORS para permitir chamadas do navegador sem bloqueio const proxyUrl = `https://corsproxy.io/?${encodeURIComponent(rawUrl)}`; try { let response = await fetch(proxyUrl); // Tentativa de fallback caso o proxy primário falhe if (!response.ok) { const fallbackProxy = `https://api.allorigins.win/raw?url=${encodeURIComponent(rawUrl)}`; response = await fetch(fallbackProxy); } if (!response.ok) { throw new Error(`Erro na API (${response.status}): Verifique o token e as permissões.`); } const data = await response.json(); if (!data || data.length === 0) { statusEl.innerHTML = `❌ Chamado #${ticketId} não encontrado.`; return; } const ticket = data[0]; // Preenche os campos do formulário automaticamente document.getElementById('ticketId').value = ticket.id || ticketId; document.getElementById('ticketSubject').value = ticket.subject || ''; document.getElementById('ticketCategory').value = ticket.category || ticket.serviceFirstLevel || 'Não informada'; document.getElementById('agentName').value = ticket.owner ? ticket.owner.businessName : 'Não atribuído'; document.getElementById('clientName').value = (ticket.clients && ticket.clients.length > 0) ? ticket.clients[0].businessName : 'Não informado'; // Consolida o histórico e apontamentos do chamado para a análise da IA let fullDescription = `ASSUNTO: ${ticket.subject}\n\n`; if (ticket.actions && ticket.actions.length > 0) { fullDescription += "HISTÓRICO / APONTAMENTOS DO CHAMADO:\n"; ticket.actions.forEach((act, idx) => { fullDescription += `\n--- Apontamento #${idx + 1} (${act.createdDate || 'Data N/I'}) ---\n${act.description || ''}\n`; }); } document.getElementById('ticketContent').value = fullDescription; statusEl.innerHTML = `✅ Chamado #${ticketId} carregado com sucesso!`; } catch (err) { console.error("Erro na busca Movidesk:", err); statusEl.innerHTML = `❌ Erro ao puxar chamado: ${err.message}. Tente colar o texto manualmente.`; } }
